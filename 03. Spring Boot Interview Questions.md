ENTITY : for creating the entities

@Entity
@Table

@Id
@GeneratedValue(strategy = GenerationType.IDENTITY)
private Long id;

@Column(nullable = false)
private String name;

@OneToMany, @ManyToOne, @OneToOne, @ManyToMany
@ManyToOne(fetch = FetchType.LAZY)
@JoinColumn(name = "user_id")
private User user;
