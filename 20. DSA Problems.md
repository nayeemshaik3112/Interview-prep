 * Q1. SECOND LARGEST ELEMENT IN AN ARRAY
 * -------------------------------------------------------------------------------------
 * Problem        : Find the second largest distinct element in an integer array.
 * Expectation    : Single pass, O(n), no sorting.
 * Brute Force    : Sort the array DESC, scan until you find element != arr[0]. O(n log n).
 * Optimized      : Maintain `first` and `second` while iterating once.
 *
 * Common Mistake : Returning arr[n-2] after sort (fails when duplicates exist).
 * Edge Cases     : All equal elements -> no second largest; array length < 2.
 *
 * Follow-ups     : - Find Kth largest without sorting? (PriorityQueue of size K)
 *                  - Second smallest? (mirror logic)

public static int secondLargest(int[] arr) {
    if (arr == null || arr.length < 2) return -1;
    int first = Integer.MIN_VALUE, second = Integer.MIN_VALUE;
    for (int n : arr) {
        if (n > first) { second = first; first = n; }
        else if (n > second && n != first) { second = n; }
    }
    return second == Integer.MIN_VALUE ? -1 : second;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q2. FIND THE DUPLICATE NUMBER  (numbers in range [1, n], array size n+1)
 * -------------------------------------------------------------------------------------
 * Problem     : Given arr of n+1 ints, each in [1..n], find ONE duplicate.
 * Expectation : O(n) time, ideally O(1) space.
 * Brute       : Two loops -> O(n^2).
 * Better      : HashSet -> O(n) time, O(n) space.
 * Optimal     : Floyd's cycle detection (treat array as linked list). O(n) / O(1).
 *               (Floyd is "medium-hard" - mention it but HashSet is acceptable.)
 *
 * Edge Cases  : Array null / length 1; multiple duplicates -> return any one.
 * Follow-ups  : Find ALL duplicates? (use sign-marking trick at index arr[i]-1)

public static int findDuplicate(int[] arr) {
    java.util.Set<Integer> seen = new java.util.HashSet<>();
    for (int n : arr) {
        if (!seen.add(n)) return n;
    }
    return -1;
}
// Time: O(n)   Space: O(n)   (Mention Floyd's for O(1) follow-up)


-------------------------------------------------------------------------------------
 * Q3. REMOVE DUPLICATES FROM SORTED ARRAY (in-place)
 * -------------------------------------------------------------------------------------
 * Problem     : Modify input so first k elements are unique; return k.
 * Approach    : Two-pointer. `i` tracks unique position; `j` scans.
 *
 * Pitfall     : Trying to "delete" inside array (Java arrays are fixed size).
 * Follow-ups  : Allow each element at most TWICE? (introduce a counter).

public static int removeDuplicatesSorted(int[] arr) {
    if (arr.length == 0) return 0;
    int i = 0;
    for (int j = 1; j < arr.length; j++) {
        if (arr[j] != arr[i]) {
            i++;
            arr[i] = arr[j];
        }
    }
    return i + 1;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q4. ROTATE ARRAY BY K POSITIONS (right rotation)
 * -------------------------------------------------------------------------------------
 * Problem     : Rotate arr to the right by k steps.
 * Brute       : Rotate one step k times -> O(n*k).
 * Better      : Use extra array of size n -> O(n) time, O(n) space.
 * Optimal     : REVERSAL TRICK -> O(n) time, O(1) space.
 *                 1) Reverse full array
 *                 2) Reverse first k
 *                 3) Reverse remaining n-k
 *
 * Pitfall     : Forgetting `k = k % n` when k > n.
 * Follow-ups  : Left rotation? (same logic, reverse first n-k then last k then all)

public static void rotateRight(int[] arr, int k) {
    int n = arr.length;
    if (n == 0) return;
    k = k % n;
    reverse(arr, 0, n - 1);
    reverse(arr, 0, k - 1);
    reverse(arr, k, n - 1);
}
private static void reverse(int[] a, int l, int r) {
    while (l < r) { int t = a[l]; a[l++] = a[r]; a[r--] = t; }
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q5. MOVE ZEROS TO END  (preserve order of non-zero elements)
 * -------------------------------------------------------------------------------------
 * Approach    : Two-pointer. `writeIdx` writes non-zeros first; then fill zeros.
 * Pitfall     : Doing it in two passes is fine; using extra array loses points.
 * Follow-ups  : Move all "negatives" to one side? (similar two-pointer).

public static void moveZeros(int[] arr) {
    int write = 0;
    for (int n : arr) if (n != 0) arr[write++] = n;
    while (write < arr.length) arr[write++] = 0;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q6. MERGE TWO SORTED ARRAYS
 * -------------------------------------------------------------------------------------
 * Problem     : Merge nums1 and nums2 (sorted) into a single sorted array.
 * Optimal     : Two-pointer merge, O(m+n).
 *
 * Follow-ups  : Merge IN-PLACE where nums1 has extra space at the end?
 *               -> Fill from the BACK using three pointers.

public static int[] mergeSorted(int[] a, int[] b) {
    int i = 0, j = 0, k = 0;
    int[] res = new int[a.length + b.length];
    while (i < a.length && j < b.length) {
        res[k++] = (a[i] <= b[j]) ? a[i++] : b[j++];
    }
    while (i < a.length) res[k++] = a[i++];
    while (j < b.length) res[k++] = b[j++];
    return res;
}
// Time: O(m+n)   Space: O(m+n)


-------------------------------------------------------------------------------------
 * Q7. FIND MIN AND MAX IN A SINGLE PASS
 * -------------------------------------------------------------------------------------
 * Trivia      : Asked to test attention to detail. Initialize from arr[0], NOT 0.

public static int[] minMax(int[] arr) {
    int min = arr[0], max = arr[0];
    for (int n : arr) {
        if (n < min) min = n;
        else if (n > max) max = n;
    }
    return new int[]{min, max};
}


-------------------------------------------------------------------------------------
 * Q8. KADANE'S ALGORITHM - MAXIMUM SUBARRAY SUM
 * -------------------------------------------------------------------------------------
 * Problem     : Find contiguous subarray with the largest sum.
 * Idea        : At each index, either EXTEND previous subarray or START fresh.
 *
 * Pitfall     : Initializing maxSoFar = 0 fails when ALL numbers are negative.
 * Follow-ups  : Return the SUBARRAY itself (track start/end indices).
 *               Maximum product subarray? (track min and max because of negatives).

public static int maxSubArraySum(int[] arr) {
    int maxSoFar = arr[0], curr = arr[0];
    for (int i = 1; i < arr.length; i++) {
        curr = Math.max(arr[i], curr + arr[i]);
        maxSoFar = Math.max(maxSoFar, curr);
    }
    return maxSoFar;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q9. SUBARRAY WITH GIVEN SUM  (positive numbers only)
 * -------------------------------------------------------------------------------------
 * Approach    : Sliding window (since all positives, sum grows monotonically).
 * Pitfall     : If negatives are allowed, sliding window FAILS - use prefix sum + HashMap.

public static int[] subarrayWithSum(int[] arr, int target) {
    int left = 0, sum = 0;
    for (int right = 0; right < arr.length; right++) {
        sum += arr[right];
        while (sum > target && left <= right) sum -= arr[left++];
        if (sum == target) return new int[]{left, right};
    }
    return new int[]{-1, -1};
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q10. LONGEST SUBARRAY WITH SUM K  (handles negatives - prefix sum + HashMap)
 * -------------------------------------------------------------------------------------
 * Idea: Store the EARLIEST occurrence of each prefix sum. If (prefix - k) exists in
 *       the map, the subarray between those two indices sums to k.
 *
 * Edge Case   : Put (0, -1) in the map up front to handle subarrays starting at 0.

public static int longestSubarraySumK(int[] arr, int k) {
    java.util.Map<Integer,Integer> map = new java.util.HashMap<>();
    map.put(0, -1);
    int prefix = 0, best = 0;
    for (int i = 0; i < arr.length; i++) {
        prefix += arr[i];
        if (map.containsKey(prefix - k))
            best = Math.max(best, i - map.get(prefix - k));
        map.putIfAbsent(prefix, i);   // earliest index only
    }
    return best;
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q11. MAJORITY ELEMENT  (appears > n/2 times) - Boyer-Moore Voting
 * -------------------------------------------------------------------------------------
 * Idea        : Count "votes". Element with the most votes survives.
 * Pitfall     : Verify the candidate in a second pass if existence is not guaranteed.

public static int majorityElement(int[] arr) {
    int count = 0, candidate = 0;
    for (int n : arr) {
        if (count == 0) candidate = n;
        count += (n == candidate) ? 1 : -1;
    }
    return candidate;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q12. MISSING NUMBER  (array contains numbers from 0..n, one missing)
 * -------------------------------------------------------------------------------------
 * Approach 1  : Sum formula  n*(n+1)/2  - actualSum.
 * Approach 2  : XOR all indices + all values - missing number remains.
 * Pitfall     : Integer overflow for large n -> use long for the sum.

public static int missingNumber(int[] arr) {
    int n = arr.length, xor = n;        // include n in xor
    for (int i = 0; i < n; i++) xor ^= i ^ arr[i];
    return xor;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q13. SORT COLORS  (Dutch National Flag - 0s, 1s, 2s)
 * -------------------------------------------------------------------------------------
 * Idea        : Three pointers - low, mid, high. Single pass.
 * Why asked   : Tests two-pointer + sorting intuition without using Arrays.sort.

public static void sortColors(int[] arr) {
    int low = 0, mid = 0, high = arr.length - 1;
    while (mid <= high) {
        switch (arr[mid]) {
            case 0: swap(arr, low++, mid++); break;
            case 1: mid++; break;
            case 2: swap(arr, mid, high--); break;
        }
    }
}
private static void swap(int[] a, int i, int j) { int t = a[i]; a[i] = a[j]; a[j] = t; }
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q14. BEST TIME TO BUY AND SELL STOCK  (single transaction)
 * -------------------------------------------------------------------------------------
 * Idea        : Track minimum so far; at every price, compute potential profit.
 * Follow-ups  : Unlimited transactions? -> sum every positive (prices[i]-prices[i-1]).
 *               At most 2 transactions? (medium-hard, skip for service companies).

public static int maxProfit(int[] prices) {
    int minSoFar = Integer.MAX_VALUE, best = 0;
    for (int p : prices) {
        if (p < minSoFar) minSoFar = p;
        else best = Math.max(best, p - minSoFar);
    }
    return best;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q15. PRODUCT OF ARRAY EXCEPT SELF  (no division allowed)
 * -------------------------------------------------------------------------------------
 * Idea        : Build prefix product and suffix product in TWO passes.
 *               result[i] = leftProduct[i] * rightProduct[i].
 *
 * Optimized   : Use the output array itself - O(1) extra space (excluding output).
 * Pitfall     : Cannot just divide totalProduct/arr[i] if zeros exist.

public static int[] productExceptSelf(int[] arr) {
    int n = arr.length;
    int[] res = new int[n];
    res[0] = 1;
    for (int i = 1; i < n; i++) res[i] = res[i-1] * arr[i-1];
    int right = 1;
    for (int i = n - 1; i >= 0; i--) {
        res[i] *= right;
        right *= arr[i];
    }
    return res;
}
// Time: O(n)   Space: O(1) extra


-------------------------------------------------------------------------------------
 * Q16. FIND PAIR WITH GIVEN SUM IN UNSORTED ARRAY  (Two Sum classic)
 * -------------------------------------------------------------------------------------
 * Brute       : O(n^2) nested loops.
 * Optimal     : HashMap of (complement -> index). Single pass O(n).
 * Follow-ups  : Return ALL pairs / count of pairs / pair indices.

public static int[] twoSum(int[] arr, int target) {
    java.util.Map<Integer,Integer> seen = new java.util.HashMap<>();
    for (int i = 0; i < arr.length; i++) {
        int need = target - arr[i];
        if (seen.containsKey(need)) return new int[]{seen.get(need), i};
        seen.put(arr[i], i);
    }
    return new int[]{-1, -1};
}
// Time: O(n)   Space: O(n)

-------------------------------------------------------------------------------------
 * Q17. REVERSE A STRING  (in place using char[])
 * -------------------------------------------------------------------------------------
 * Pitfall     : Strings are immutable in Java - cannot reverse "in place" directly.
 *               Convert to char[] first, OR use StringBuilder.reverse().

public static String reverseString(String s) {
    char[] c = s.toCharArray();
    int l = 0, r = c.length - 1;
    while (l < r) { char t = c[l]; c[l++] = c[r]; c[r--] = t; }
    return new String(c);
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q18. CHECK PALINDROME  (alphanumeric only, ignore case) - classic clarifier question
 * -------------------------------------------------------------------------------------
 * ALWAYS ASK : "Should we ignore non-alphanumeric chars? Case sensitive?"
 * Interview tip: the clarifying question itself earns points.

public static boolean isPalindrome(String s) {
    int l = 0, r = s.length() - 1;
    while (l < r) {
        while (l < r && !Character.isLetterOrDigit(s.charAt(l))) l++;
        while (l < r && !Character.isLetterOrDigit(s.charAt(r))) r--;
        if (Character.toLowerCase(s.charAt(l++)) != Character.toLowerCase(s.charAt(r--)))
            return false;
    }
    return true;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q19. CHECK ANAGRAM
 * -------------------------------------------------------------------------------------
 * Approach 1  : Sort both strings and compare - O(n log n).
 * Approach 2  : Frequency array of size 26 (lowercase) - O(n) time, O(1) space.
 *
 * Pitfall     : Unicode strings -> use HashMap<Character,Integer> instead of int[26].

public static boolean isAnagram(String a, String b) {
    if (a.length() != b.length()) return false;
    int[] freq = new int[26];
    for (int i = 0; i < a.length(); i++) {
        freq[a.charAt(i) - 'a']++;
        freq[b.charAt(i) - 'a']--;
    }
    for (int f : freq) if (f != 0) return false;
    return true;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q20. FIRST NON-REPEATING CHARACTER
 * -------------------------------------------------------------------------------------
 * Idea        : Two passes - first build frequency map, second find first char with
 *               count == 1. Preserves the ORIGINAL order.
 *
 * Pitfall     : Using HashMap iteration to find "first" - order not guaranteed
 *               (use LinkedHashMap, or iterate the input string).

public static char firstNonRepeating(String s) {
    int[] freq = new int[256];
    for (char c : s.toCharArray()) freq[c]++;
    for (char c : s.toCharArray()) if (freq[c] == 1) return c;
    return '_';
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q21. LONGEST SUBSTRING WITHOUT REPEATING CHARACTERS  (CLASSIC SLIDING WINDOW)
 * -------------------------------------------------------------------------------------
 * Idea: Expand right pointer; when duplicate found, jump left to (lastIndex + 1).
 * Pitfall: Make sure `left` only moves FORWARD: left = Math.max(left, map.get(c)+1).
 *
 * Follow-ups: Return the substring itself; allow at most K distinct chars.

public static int lengthOfLongestSubstring(String s) {
    java.util.Map<Character,Integer> map = new java.util.HashMap<>();
    int best = 0, left = 0;
    for (int right = 0; right < s.length(); right++) {
        char c = s.charAt(right);
        if (map.containsKey(c)) left = Math.max(left, map.get(c) + 1);
        map.put(c, right);
        best = Math.max(best, right - left + 1);
    }
    return best;
}
// Time: O(n)   Space: O(min(n, charset))


-------------------------------------------------------------------------------------
 * Q22. FREQUENCY COUNT OF CHARACTERS  (foundation pattern)
 * -------------------------------------------------------------------------------------
 * Used as a SUB-ROUTINE in: anagram, first non-repeating, group anagrams, etc.

public static java.util.Map<Character,Integer> charFrequency(String s) {
    java.util.Map<Character,Integer> map = new java.util.HashMap<>();
    for (char c : s.toCharArray()) map.merge(c, 1, Integer::sum);
    return map;
}


-------------------------------------------------------------------------------------
 * Q23. FIND ALL DUPLICATE CHARACTERS IN A STRING
 * -------------------------------------------------------------------------------------
 * Approach    : Frequency map; collect chars with count > 1.
 * Pitfall     : Preserve insertion order if the interviewer asks - use LinkedHashMap.

public static java.util.List<Character> duplicateChars(String s) {
    java.util.Map<Character,Integer> map = new java.util.LinkedHashMap<>();
    for (char c : s.toCharArray()) map.merge(c, 1, Integer::sum);
    java.util.List<Character> res = new java.util.ArrayList<>();
    for (var e : map.entrySet()) if (e.getValue() > 1) res.add(e.getKey());
    return res;
}


-------------------------------------------------------------------------------------
 * Q24. REVERSE WORDS IN A STRING  ("the sky is blue" -> "blue is sky the")
 * -------------------------------------------------------------------------------------
 * Pitfall     : Trim leading/trailing spaces, collapse multiple spaces.
 * Approach    : Split on \\s+, reverse, join with single space.
 *
 * Follow-ups  : Do it WITHOUT split (in-place using char[] + reversal trick).

public static String reverseWords(String s) {
    String[] tokens = s.trim().split("\\s+");
    StringBuilder sb = new StringBuilder();
    for (int i = tokens.length - 1; i >= 0; i--) {
        sb.append(tokens[i]);
        if (i > 0) sb.append(' ');
    }
    return sb.toString();
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q25. STRING COMPRESSION  ("aaabbc" -> "a3b2c1")  Run-Length style
 * -------------------------------------------------------------------------------------
 * Pitfall     : Use StringBuilder (NOT String concatenation in a loop).
 * Variant     : Return the SHORTER of original vs compressed.

public static String compress(String s) {
    StringBuilder sb = new StringBuilder();
    int n = s.length(), i = 0;
    while (i < n) {
        int j = i;
        while (j < n && s.charAt(j) == s.charAt(i)) j++;
        sb.append(s.charAt(i)).append(j - i);
        i = j;
    }
    return sb.toString();
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q26. CHECK IF TWO STRINGS ARE ROTATIONS OF EACH OTHER
 * -------------------------------------------------------------------------------------
 * Trick       : s2 is a rotation of s1 iff (s1+s1).contains(s2) AND lengths match.
 * Why elegant : Single line. Interviewers love this trick.

public static boolean areRotations(String s1, String s2) {
    return s1.length() == s2.length() && (s1 + s1).contains(s2);
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q27. LONGEST COMMON PREFIX  among an array of strings
 * -------------------------------------------------------------------------------------
 * Approach    : Vertical scan - compare char by char across all strings.
 * Edge Case   : Empty array -> return "".

public static String longestCommonPrefix(String[] strs) {
    if (strs == null || strs.length == 0) return "";
    for (int i = 0; i < strs[0].length(); i++) {
        char c = strs[0].charAt(i);
        for (int j = 1; j < strs.length; j++) {
            if (i == strs[j].length() || strs[j].charAt(i) != c)
                return strs[0].substring(0, i);
        }
    }
    return strs[0];
}
// Time: O(N*M)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q28. VALID PARENTHESES  ("()[]{}" -> true, "(]" -> false)
 * -------------------------------------------------------------------------------------
 * Classic Stack problem.  Push opens, pop on close and verify match.
 * Pitfall     : Don't forget to check if stack is EMPTY at the end.

public static boolean isValidBrackets(String s) {
    java.util.Deque<Character> st = new java.util.ArrayDeque<>();
    for (char c : s.toCharArray()) {
        if (c == '(' || c == '[' || c == '{') st.push(c);
        else {
            if (st.isEmpty()) return false;
            char top = st.pop();
            if ((c == ')' && top != '(') ||
                (c == ']' && top != '[') ||
                (c == '}' && top != '{')) return false;
        }
    }
    return st.isEmpty();
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q29. GROUP ANAGRAMS
 * -------------------------------------------------------------------------------------
 * Idea: Use sorted version of each string as a HashMap key.
 * Alternative key: a "26-int count string" - O(n*k) instead of O(n*k log k).

public static java.util.List<java.util.List<String>> groupAnagrams(String[] strs) {
    java.util.Map<String, java.util.List<String>> map = new java.util.HashMap<>();
    for (String s : strs) {
        char[] c = s.toCharArray();
        java.util.Arrays.sort(c);
        map.computeIfAbsent(new String(c), k -> new java.util.ArrayList<>()).add(s);
    }
    return new java.util.ArrayList<>(map.values());
}
// Time: O(n * k log k)   Space: O(n * k)

-------------------------------------------------------------------------------------
 * Q30. CONTAINS DUPLICATE
 * -------------------------------------------------------------------------------------
 * HashSet "add" returns false on duplicate - shortest possible solution.

public static boolean containsDuplicate(int[] arr) {
    java.util.Set<Integer> set = new java.util.HashSet<>();
    for (int n : arr) if (!set.add(n)) return true;
    return false;
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q31. CONTAINS DUPLICATE WITHIN K DISTANCE  (sliding HashSet)
 * -------------------------------------------------------------------------------------
 * Idea: Keep a window of size k as a HashSet; check membership before adding;
 *       remove the element going out of the window.

public static boolean containsNearbyDuplicate(int[] arr, int k) {
    java.util.Set<Integer> window = new java.util.HashSet<>();
    for (int i = 0; i < arr.length; i++) {
        if (!window.add(arr[i])) return true;
        if (window.size() > k) window.remove(arr[i - k]);
    }
    return false;
}
// Time: O(n)   Space: O(k)


-------------------------------------------------------------------------------------
 * Q32. SUBARRAY SUM EQUALS K  (count, with negatives allowed)  -> PREFIX SUM + MAP
 * -------------------------------------------------------------------------------------
 * Classic pattern. Map stores (prefixSum -> howManyTimes it occurred).
 * Initialize with (0 -> 1) to count subarrays starting from index 0.

public static int subarraySumEqualsK(int[] arr, int k) {
    java.util.Map<Integer,Integer> map = new java.util.HashMap<>();
    map.put(0, 1);
    int prefix = 0, count = 0;
    for (int n : arr) {
        prefix += n;
        count += map.getOrDefault(prefix - k, 0);
        map.merge(prefix, 1, Integer::sum);
    }
    return count;
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q33. LONGEST CONSECUTIVE SEQUENCE  ([100,4,200,1,3,2] -> 4 for [1,2,3,4])
 * -------------------------------------------------------------------------------------
 * Trick       : Put all in a HashSet. Start counting ONLY when current-1 not in set
 *               (i.e., current is the start of a streak). This makes it O(n).
 *
 * Common Mistake: O(n log n) by sorting - acceptable but interviewers want O(n).

public static int longestConsecutive(int[] arr) {
    java.util.Set<Integer> set = new java.util.HashSet<>();
    for (int n : arr) set.add(n);
    int best = 0;
    for (int n : set) {
        if (!set.contains(n - 1)) {
            int len = 1, cur = n;
            while (set.contains(cur + 1)) { cur++; len++; }
            best = Math.max(best, len);
        }
    }
    return best;
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q34. INTERSECTION OF TWO ARRAYS  (unique elements)
 * -------------------------------------------------------------------------------------
 * Build set from smaller array; check membership while scanning the other.

public static int[] intersection(int[] a, int[] b) {
    java.util.Set<Integer> set = new java.util.HashSet<>();
    for (int n : a) set.add(n);
    java.util.Set<Integer> res = new java.util.HashSet<>();
    for (int n : b) if (set.contains(n)) res.add(n);
    return res.stream().mapToInt(Integer::intValue).toArray();
}


-------------------------------------------------------------------------------------
 * Q35. FIND ALL DUPLICATES IN AN ARRAY  (values in [1..n], O(1) extra space)
 * -------------------------------------------------------------------------------------
 * Trick       : Mark visited by negating arr[abs(arr[i]) - 1]. If already negative,
 *               it's a duplicate.
 * Caveat      : Modifies the input - ask interviewer if that's allowed.

public static java.util.List<Integer> findAllDuplicates(int[] arr) {
    java.util.List<Integer> res = new java.util.ArrayList<>();
    for (int i = 0; i < arr.length; i++) {
        int idx = Math.abs(arr[i]) - 1;
        if (arr[idx] < 0) res.add(idx + 1);
        else arr[idx] = -arr[idx];
    }
    return res;
}
// Time: O(n)   Space: O(1) extra


-------------------------------------------------------------------------------------
 * Q36. TOP K FREQUENT ELEMENTS  (bucket-sort style without DP/heap-heavy code)
 * -------------------------------------------------------------------------------------
 * Approach 1  : Frequency map + min-heap of size K -> O(n log k).
 * Approach 2  : Bucket sort by frequency -> O(n).

public static int[] topKFrequent(int[] arr, int k) {
    java.util.Map<Integer,Integer> freq = new java.util.HashMap<>();
    for (int n : arr) freq.merge(n, 1, Integer::sum);
    java.util.PriorityQueue<int[]> pq =
        new java.util.PriorityQueue<>((x, y) -> x[1] - y[1]);
    for (var e : freq.entrySet()) {
        pq.offer(new int[]{e.getKey(), e.getValue()});
        if (pq.size() > k) pq.poll();
    }
    int[] res = new int[k];
    for (int i = k - 1; i >= 0; i--) res[i] = pq.poll()[0];
    return res;
}
// Time: O(n log k)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q37. ISOMORPHIC STRINGS  ("egg" and "add" -> true; "foo" and "bar" -> false)
 * -------------------------------------------------------------------------------------
 * Two-way mapping check using two HashMaps (or two int arrays of size 256).

public static boolean isIsomorphic(String s, String t) {
    if (s.length() != t.length()) return false;
    int[] mapS = new int[256], mapT = new int[256];
    for (int i = 0; i < s.length(); i++) {
        if (mapS[s.charAt(i)] != mapT[t.charAt(i)]) return false;
        mapS[s.charAt(i)] = i + 1;
        mapT[t.charAt(i)] = i + 1;
    }
    return true;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q38. HAPPY NUMBER  (replace n by sum of squares of digits; ends at 1 or cycles)
 * -------------------------------------------------------------------------------------
 * Use HashSet to detect cycles. Asked at Wells Fargo / Accenture occasionally.

public static boolean isHappy(int n) {
    java.util.Set<Integer> seen = new java.util.HashSet<>();
    while (n != 1 && seen.add(n)) {
        int sum = 0;
        while (n > 0) { int d = n % 10; sum += d * d; n /= 10; }
        n = sum;
    }
    return n == 1;
}


-------------------------------------------------------------------------------------
 * Q39. FOUR SUM COUNT  (count tuples (i,j,k,l) with A[i]+B[j]+C[k]+D[l] == 0)
 * -------------------------------------------------------------------------------------
 * Idea: Reduce to two-sum on (A+B) and -(C+D). Classic HashMap trick.
 * (Slightly medium-hard; mention as a follow-up to Two Sum.)

public static int fourSumCount(int[] A, int[] B, int[] C, int[] D) {
    java.util.Map<Integer,Integer> map = new java.util.HashMap<>();
    for (int a : A) for (int b : B) map.merge(a + b, 1, Integer::sum);
    int count = 0;
    for (int c : C) for (int d : D) count += map.getOrDefault(-(c + d), 0);
    return count;
}
// Time: O(n^2)   Space: O(n^2)

-------------------------------------------------------------------------------------
 * Q40. MAXIMUM SUM SUBARRAY OF SIZE K  (FIXED WINDOW)
 * -------------------------------------------------------------------------------------
 * Idea: Compute sum of first K. Slide by ADDING arr[i] and SUBTRACTING arr[i-k].
 * The foundational fixed-window problem - memorize this template.

public static int maxSumWindowK(int[] arr, int k) {
    int sum = 0;
    for (int i = 0; i < k; i++) sum += arr[i];
    int best = sum;
    for (int i = k; i < arr.length; i++) {
        sum += arr[i] - arr[i - k];
        best = Math.max(best, sum);
    }
    return best;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q41. LONGEST SUBSTRING WITH AT MOST K DISTINCT CHARACTERS  (VARIABLE WINDOW)
 * -------------------------------------------------------------------------------------
 * Template for ALL "at most K of something" problems.
 * Shrink window from the LEFT while invariant (distinct > k) is violated.

public static int longestSubstringKDistinct(String s, int k) {
    java.util.Map<Character,Integer> map = new java.util.HashMap<>();
    int left = 0, best = 0;
    for (int right = 0; right < s.length(); right++) {
        map.merge(s.charAt(right), 1, Integer::sum);
        while (map.size() > k) {
            char lc = s.charAt(left++);
            if (map.merge(lc, -1, Integer::sum) == 0) map.remove(lc);
        }
        best = Math.max(best, right - left + 1);
    }
    return best;
}
// Time: O(n)   Space: O(k)


-------------------------------------------------------------------------------------
 * Q42. MINIMUM WINDOW SUBSTRING  (medium-hard - asked at Wells Fargo / product cos)
 * -------------------------------------------------------------------------------------
 * Find the smallest substring in s that contains all chars of t (with frequencies).
 * Template = expand right, shrink left while "have == need".

public static String minWindow(String s, String t) {
    if (s.length() < t.length()) return "";
    int[] need = new int[256];
    for (char c : t.toCharArray()) need[c]++;
    int required = t.length(), left = 0, bestLen = Integer.MAX_VALUE, bestL = 0;
    for (int right = 0; right < s.length(); right++) {
        if (need[s.charAt(right)]-- > 0) required--;
        while (required == 0) {
            if (right - left + 1 < bestLen) { bestLen = right - left + 1; bestL = left; }
            if (++need[s.charAt(left++)] > 0) required++;
        }
    }
    return bestLen == Integer.MAX_VALUE ? "" : s.substring(bestL, bestL + bestLen);
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q43. FIND ALL ANAGRAMS IN A STRING  (return starting indices)
 * -------------------------------------------------------------------------------------
 * Fixed window of size p.length(). Compare frequency arrays.

public static java.util.List<Integer> findAnagrams(String s, String p) {
    java.util.List<Integer> res = new java.util.ArrayList<>();
    if (s.length() < p.length()) return res;
    int[] pf = new int[26], sf = new int[26];
    for (char c : p.toCharArray()) pf[c - 'a']++;
    for (int i = 0; i < s.length(); i++) {
        sf[s.charAt(i) - 'a']++;
        if (i >= p.length()) sf[s.charAt(i - p.length()) - 'a']--;
        if (java.util.Arrays.equals(pf, sf)) res.add(i - p.length() + 1);
    }
    return res;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q44. LONGEST REPEATING CHARACTER REPLACEMENT  (asked at IBM / Deloitte sometimes)
 * -------------------------------------------------------------------------------------
 * Within window: (windowLength - countOfMostFrequentChar) <= k.
 * If violated, shrink left.

public static int characterReplacement(String s, int k) {
    int[] freq = new int[26];
    int left = 0, maxFreq = 0, best = 0;
    for (int right = 0; right < s.length(); right++) {
        maxFreq = Math.max(maxFreq, ++freq[s.charAt(right) - 'A']);
        if (right - left + 1 - maxFreq > k) freq[s.charAt(left++) - 'A']--;
        best = Math.max(best, right - left + 1);
    }
    return best;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q45. PERMUTATION IN STRING  (does s2 contain any permutation of s1?)
 * -------------------------------------------------------------------------------------
 * Same idea as "Find Anagrams" but returns boolean.

public static boolean checkInclusion(String s1, String s2) {
    if (s2.length() < s1.length()) return false;
    int[] f1 = new int[26], f2 = new int[26];
    for (char c : s1.toCharArray()) f1[c - 'a']++;
    for (int i = 0; i < s2.length(); i++) {
        f2[s2.charAt(i) - 'a']++;
        if (i >= s1.length()) f2[s2.charAt(i - s1.length()) - 'a']--;
        if (java.util.Arrays.equals(f1, f2)) return true;
    }
    return false;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q46. SMALLEST SUBARRAY WITH SUM >= TARGET  (positive numbers)
 * -------------------------------------------------------------------------------------
 * Variable window: shrink left while sum still >= target. Track min length.

public static int minSubArrayLen(int target, int[] arr) {
    int left = 0, sum = 0, best = Integer.MAX_VALUE;
    for (int right = 0; right < arr.length; right++) {
        sum += arr[right];
        while (sum >= target) {
            best = Math.min(best, right - left + 1);
            sum -= arr[left++];
        }
    }
    return best == Integer.MAX_VALUE ? 0 : best;
}
// Time: O(n)   Space: O(1)

-------------------------------------------------------------------------------------
 * Q47. PAIR SUM IN A SORTED ARRAY
 * -------------------------------------------------------------------------------------
 * Two pointers from both ends. If sum < target -> l++, else r--.
 * Beats HashMap on space (O(1)).

public static int[] pairSumSorted(int[] arr, int target) {
    int l = 0, r = arr.length - 1;
    while (l < r) {
        int sum = arr[l] + arr[r];
        if (sum == target) return new int[]{l, r};
        else if (sum < target) l++;
        else r--;
    }
    return new int[]{-1, -1};
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q48. THREE SUM  (all unique triplets summing to 0)
 * -------------------------------------------------------------------------------------
 * Pattern     : Sort, fix one element, two-pointer on the rest. Skip duplicates.
 *
 * Common Mistake: Forgetting to skip duplicates for BOTH the fixed element AND both
 *                 inner pointers (causes duplicate triplets in the result).

public static java.util.List<java.util.List<Integer>> threeSum(int[] arr) {
    java.util.Arrays.sort(arr);
    java.util.List<java.util.List<Integer>> res = new java.util.ArrayList<>();
    for (int i = 0; i < arr.length - 2; i++) {
        if (i > 0 && arr[i] == arr[i - 1]) continue;
        int l = i + 1, r = arr.length - 1;
        while (l < r) {
            int sum = arr[i] + arr[l] + arr[r];
            if (sum == 0) {
                res.add(java.util.Arrays.asList(arr[i], arr[l], arr[r]));
                while (l < r && arr[l] == arr[l + 1]) l++;
                while (l < r && arr[r] == arr[r - 1]) r--;
                l++; r--;
            } else if (sum < 0) l++;
            else r--;
        }
    }
    return res;
}
// Time: O(n^2)   Space: O(1) (excluding output)


-------------------------------------------------------------------------------------
 * Q49. CONTAINER WITH MOST WATER
 * -------------------------------------------------------------------------------------
 * Greedy two-pointer: always move the SHORTER side inward (others can only shrink).

public static int maxArea(int[] height) {
    int l = 0, r = height.length - 1, best = 0;
    while (l < r) {
        int h = Math.min(height[l], height[r]);
        best = Math.max(best, h * (r - l));
        if (height[l] < height[r]) l++; else r--;
    }
    return best;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q50. TRAPPING RAIN WATER  (medium-hard - asked at Wells Fargo & product companies)
 * -------------------------------------------------------------------------------------
 * Two-pointer with leftMax & rightMax. Whichever side has smaller max bounds the water.
 * Beats the O(n) extra-space prefix-max / suffix-max approach.

public static int trapRainWater(int[] h) {
    int l = 0, r = h.length - 1, lm = 0, rm = 0, total = 0;
    while (l < r) {
        if (h[l] < h[r]) {
            if (h[l] >= lm) lm = h[l]; else total += lm - h[l];
            l++;
        } else {
            if (h[r] >= rm) rm = h[r]; else total += rm - h[r];
            r--;
        }
    }
    return total;
}
// Time: O(n)   Space: O(1)


-------------------------------------------------------------------------------------
 * Q51. SQUARES OF A SORTED ARRAY  (input may have negatives - already sorted)
 * -------------------------------------------------------------------------------------
 * Trick: Largest square is at one of the two ends. Fill result from the back.
 * Naive O(n log n) by sort - mention but reject. Optimal is O(n).

public static int[] sortedSquares(int[] arr) {
    int n = arr.length, l = 0, r = n - 1, k = n - 1;
    int[] res = new int[n];
    while (l <= r) {
        int ls = arr[l] * arr[l], rs = arr[r] * arr[r];
        if (ls > rs) { res[k--] = ls; l++; }
        else         { res[k--] = rs; r--; }
    }
    return res;
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q52. REMOVE ELEMENT IN-PLACE  (delete all occurrences of val; return new length)
 * -------------------------------------------------------------------------------------
 * Two-pointer write index. Cousin of "remove duplicates" and "move zeros".

public static int removeElement(int[] arr, int val) {
    int w = 0;
    for (int n : arr) if (n != val) arr[w++] = n;
    return w;
}

 * Q53. REVERSE VOWELS IN A STRING
 * -------------------------------------------------------------------------------------
 * Two pointers; advance only when both ends point at vowels; swap.

public static String reverseVowels(String s) {
    char[] c = s.toCharArray();
    int l = 0, r = c.length - 1;
    String vowels = "aeiouAEIOU";
    while (l < r) {
        while (l < r && vowels.indexOf(c[l]) < 0) l++;
        while (l < r && vowels.indexOf(c[r]) < 0) r--;
        char t = c[l]; c[l++] = c[r]; c[r--] = t;
    }
    return new String(c);
}

-------------------------------------------------------------------------------------
 * Q54. NEXT GREATER ELEMENT  (for each index, find the next bigger one to its right)
 * -------------------------------------------------------------------------------------
 * Monotonic decreasing stack. While stack top < current, pop and assign current.
 * Pattern used in: Daily Temperatures, Stock Span, etc.

public static int[] nextGreaterElement(int[] arr) {
    int n = arr.length;
    int[] res = new int[n];
    java.util.Arrays.fill(res, -1);
    java.util.Deque<Integer> st = new java.util.ArrayDeque<>();   // stack of indices
    for (int i = 0; i < n; i++) {
        while (!st.isEmpty() && arr[st.peek()] < arr[i])
            res[st.pop()] = arr[i];
        st.push(i);
    }
    return res;
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q55. IMPLEMENT QUEUE USING TWO STACKS  (classic OOP-style question)
 * -------------------------------------------------------------------------------------
 * Idea: One stack for enqueue, another for dequeue.
 *       Move elements from in -> out ONLY when out is empty (amortized O(1)).

public static class MyQueue {
    private final java.util.Deque<Integer> in  = new java.util.ArrayDeque<>();
    private final java.util.Deque<Integer> out = new java.util.ArrayDeque<>();
    public void enqueue(int x) { in.push(x); }
    public int dequeue() {
        if (out.isEmpty()) while (!in.isEmpty()) out.push(in.pop());
        return out.pop();
    }
    public int peek() {
        if (out.isEmpty()) while (!in.isEmpty()) out.push(in.pop());
        return out.peek();
    }
    public boolean isEmpty() { return in.isEmpty() && out.isEmpty(); }
}


-------------------------------------------------------------------------------------
 * Q56. IMPLEMENT STACK USING TWO QUEUES
 * -------------------------------------------------------------------------------------
 * Approach: On push, rotate the queue so the latest element is at the front.

public static class MyStack {
    private final java.util.Deque<Integer> q = new java.util.ArrayDeque<>();
    public void push(int x) {
        q.offer(x);
        int n = q.size();
        while (n-- > 1) q.offer(q.poll());        // rotate
    }
    public int pop()  { return q.poll(); }
    public int top()  { return q.peek(); }
    public boolean empty() { return q.isEmpty(); }
}


-------------------------------------------------------------------------------------
 * Q57. MIN STACK  (push, pop, top, getMin - all O(1))
 * -------------------------------------------------------------------------------------
 * Trick: Maintain a parallel "min stack". Push to min stack only when new value <= current min.
 *        On pop, pop min stack too if popped value equals current min.

public static class MinStack {
    private final java.util.Deque<Integer> stack    = new java.util.ArrayDeque<>();
    private final java.util.Deque<Integer> minStack = new java.util.ArrayDeque<>();
    public void push(int x) {
        stack.push(x);
        if (minStack.isEmpty() || x <= minStack.peek()) minStack.push(x);
    }
    public void pop() {
        int v = stack.pop();
        if (v == minStack.peek()) minStack.pop();
    }
    public int top()    { return stack.peek(); }
    public int getMin() { return minStack.peek(); }
}


-------------------------------------------------------------------------------------
 * Q58. DAILY TEMPERATURES  (days until a warmer temperature for each day)
 * -------------------------------------------------------------------------------------
 * Monotonic stack of indices. For each new day, pop colder days and write the gap.

public static int[] dailyTemperatures(int[] t) {
    int n = t.length;
    int[] res = new int[n];
    java.util.Deque<Integer> st = new java.util.ArrayDeque<>();
    for (int i = 0; i < n; i++) {
        while (!st.isEmpty() && t[st.peek()] < t[i]) {
            int idx = st.pop();
            res[idx] = i - idx;
        }
        st.push(i);
    }
    return res;
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q59. EVALUATE REVERSE POLISH NOTATION  ("2 1 + 3 *" = 9)
 * -------------------------------------------------------------------------------------
 * Stack of integers. On operator, pop b then a, compute (a op b), push.
 * Pitfall: Order matters for - and /  =>  a - b, a / b (NOT b - a).

public static int evalRPN(String[] tokens) {
    java.util.Deque<Integer> st = new java.util.ArrayDeque<>();
    for (String tok : tokens) {
        switch (tok) {
            case "+": case "-": case "*": case "/":
                int b = st.pop(), a = st.pop();
                switch (tok) {
                    case "+": st.push(a + b); break;
                    case "-": st.push(a - b); break;
                    case "*": st.push(a * b); break;
                    case "/": st.push(a / b); break;
                }
                break;
            default:
                st.push(Integer.parseInt(tok));
        }
    }
    return st.pop();
}
// Time: O(n)   Space: O(n)


-------------------------------------------------------------------------------------
 * Q60. SLIDING WINDOW MAXIMUM  (medium-hard - asked at product companies)
 * -------------------------------------------------------------------------------------
 * Monotonic decreasing deque of indices. Front = max of current window.
 *
 * Steps per index i:
 *   1) Remove indices out of window from front.
 *   2) Pop from back while arr[back] <= arr[i].
 *   3) Push i.
 *   4) When i >= k-1, record arr[deque.front()].
 *
 * Skip if going for service-only roles. Master for startups/product roles.

public static int[] maxSlidingWindow(int[] arr, int k) {
    int n = arr.length;
    int[] res = new int[n - k + 1];
    java.util.Deque<Integer> dq = new java.util.ArrayDeque<>();
    for (int i = 0; i < n; i++) {
        if (!dq.isEmpty() && dq.peekFirst() <= i - k) dq.pollFirst();
        while (!dq.isEmpty() && arr[dq.peekLast()] <= arr[i]) dq.pollLast();
        dq.offerLast(i);
        if (i >= k - 1) res[i - k + 1] = arr[dq.peekFirst()];
    }
    return res;
}
// Time: O(n)   Space: O(k)

 *   - For stack: prefer Deque<Integer> with ArrayDeque - never the legacy Stack class.
 *   - For queue: ArrayDeque or LinkedList - never `new Queue<>()` (Queue is interface).
 *   - HashMap iteration order is not guaranteed - use LinkedHashMap if order matters.
 *   - Sorting an int[]: Arrays.sort(arr) - quick sort, NOT stable for objects.
 *
 *  CORE COMPLEXITIES TO MEMORIZE:
 *  ------------------------------
 *   HashMap put/get/containsKey       : O(1) avg, O(n) worst case
 *   HashSet add/contains              : O(1) avg
 *   ArrayList add (amortized)         : O(1); get/set: O(1); remove(i): O(n)
 *   LinkedList add at ends            : O(1); get(i): O(n)
 *   TreeMap / TreeSet                 : O(log n) for all
 *   Arrays.sort(int[])                : O(n log n)  dual-pivot quicksort
 *   Collections.sort(List<T>)         : O(n log n)  Timsort, stable
 *   PriorityQueue offer/poll          : O(log n);  peek: O(1)
 *   ArrayDeque push/pop/offer/poll    : O(1) amortized
 *   String.substring, String.charAt   : O(1) for charAt, O(k) for substring
